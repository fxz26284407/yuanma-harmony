cmake_minimum_required(VERSION 3.16)
project(codex_mobile_native LANGUAGES C CXX)
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.." ABSOLUTE)
set(NOISE_SOURCE "${REPO_ROOT}/.local/noise-c")
if(NOT EXISTS "${NOISE_SOURCE}/include/noise/protocol.h")
  message(FATAL_ERROR "Run npm run prepare:crypto in the repository root to provision pinned Noise-C sources")
endif()
add_subdirectory("${REPO_ROOT}/interop" "${CMAKE_CURRENT_BINARY_DIR}/noise-core")
add_library(codexnoise SHARED napi_noise.cpp)
target_compile_features(codexnoise PRIVATE cxx_std_17)
target_compile_options(codexnoise PRIVATE -fexceptions)
target_link_libraries(codexnoise PRIVATE noiseprotocol libace_napi.z.so)
